This is tight but feasible. 2B tokens on a 135M model = 6*N*D ~ 1.62 \times 10^{18} FLOPs. Over 28,800s you need ∼56 TFLOPs sustained. An RTX 5090 Blackwell is 21760 CUDA Cores with ∼1800 TFLOPS FP16 / ∼3600 FP8, so BF16 is the right target. Blackwell is sm_120 and requires CUDA >=12.8 and PyTorch/CUDA binaries built for sm_120. FlashAttention-3 officially targets Hopper sm_90a and requires CUDA >=12.3, so for 5090 you must recompile with sm_120a.

// src/data/fetcher.rs
use arrow::array::StringArray;
use futures::StreamExt;
use parquet::arrow::ParquetRecordBatchStreamBuilder;
use tokio::io::AsyncWriteExt;

pub async fn stream_stack_v2(parquet_urls: Vec<String>, tx: tokio::sync::mpsc::Sender<String>) {
    let client = reqwest::Client::new();
    for url in parquet_urls {
        let resp = client.get(&url).send().await.unwrap().bytes_stream();
        // Parquet async reader directly from stream -> no temp files
        let mut stream_reader = {
            let body = resp.map(|b| b.unwrap()).collect::<Vec<_>>().await.concat();
            ParquetRecordBatchStreamBuilder::new(bytes::Bytes::from(body)).await.unwrap().build().unwrap()
        };
        while let Some(batch) = stream_reader.next().await {
            let batch = batch.unwrap();
            let col = batch.column_by_name("content").unwrap();
            let arr = col.as_any().downcast_ref::<StringArray>().unwrap();
            for i in 0..arr.len() {
                if let Some(code) = arr.value_opt(i) { let _ = tx.send(code.to_string()).await; }
            }
        }
    }
}

// src/data/dedup.rs
use rayon::prelude::*;
use std::collections::HashSet;

fn shingles(s: &str, k: usize) -> HashSet<u64> {
    s.as_bytes().windows(k).map(|w| {
        let mut h = 0xcbf29ce484222325u64;
        for &b in w { h ^= b as u64; h = h.wrapping_mul(0x100000001b3); }
        h
    }).collect()
}

pub fn minhash_dedup(corpus: Vec<String>) -> Vec<String> {
    const N_HASH: usize = 128;
    let sigs: Vec<Vec<u64>> = corpus.par_iter().map(|s| {
        let sh = shingles(s, 5);
        (0..N_HASH).map(|i| sh.iter().map(|x| x.wrapping_add(i as u64*0x9e3779b97f4a7c15)).min().unwrap_or(0)).collect()
    }).collect();

    let mut keep = vec![true; corpus.len()];
    for i in 0..corpus.len() {
        if!keep[i] { continue; }
        for j in i+1..corpus.len() {
            if!keep[j] { continue; }
            let inter = sigs[i].iter().zip(&sigs[j]).filter(|(a,b)| a==b).count() as f32 / N_HASH as f32;
            if inter > 0.85 { keep[j]=false; }
        }
    }
    corpus.into_iter().enumerate().filter(|(i,_)| keep[*i]).map(|(_,s)| s).collect()
}

pub fn write_u16_shard(tokens: &[u16], out_path: &str) {
    let file = OpenOptions::new().read(true).write(true).create(true).open(out_path).unwrap();
    file.set_len((tokens.len()*2) as u64).unwrap();
    let mut mmap = unsafe { MmapMut::map_mut(&file).unwrap() };

    shorts.copy_from_slice(tokens); // zero-copy streaming later
    mmap.flush().unwrap();
}

        let var = (x.sqr()?.sum_keepdim(2)? / x.dim(2)? as f64)?;

    }

Rust

// src/model/gqa.rs
pub fn grouped_query_attention(q: &Tensor, k: &Tensor, v: &Tensor, n_q: usize, n_kv: usize) -> Result<Tensor> {
    // q: [B, L, 12, 64], k,v: [B, L, 4, 64]
    // repeat kv heads: 12/4=3
    let repeat = n_q / n_kv;
    let k_rep = k.repeat_interleave(repeat, 2)?; // -> [B, L, 12, 64]
    let v_rep = v.repeat_interleave(repeat, 2)?;
    // SDPA: QK^T / sqrt(d) -> softmax -> @V
    // If feature cuda, call flash_attn3_fwd_bf16 via FFI
    // else candle_core::candle_flashattn?
    let scores = (q.matmul(&k_rep.t()?)? / (q.dim(3)? as f64).sqrt())?;
    let attn = candle_nn::ops::softmax(&scores, 3)?;
    attn.matmul(&v_rep)
}

pub struct SwiGLU { gate: candle_nn::Linear, up: candle_nn::Linear, down: candle_nn::Linear }
impl SwiGLU {
    pub fn forward(&self, x: &Tensor) -> Result<Tensor> {
        let g = candle_nn::ops::silu(&self.gate.forward(x)?)?;
        let u = self.up.forward(x)?;
        self.down.forward(&(g * u)?)
    }
}

5. Pre-Training Loop & VRAM Management

Target: 70k tokens/sec. Micro-batch 32 seq * 2048 = 65536 tokens. Accum 8 => 524k tokens effective.
Rust

// src/train/loop.rs
use candle_core::{Device, DType, Tensor};

pub async fn train(cfg: Config) -> anyhow::Result<()> {
    let dev = Device::new_cuda(0)?;
    let mut model = Llama::new(&cfg, &dev)?; // bf16 params
    let mut opt = AdamW::new(model.params(), 2.5e-3, 0.9, 0.95, 0.1, DType::BF16)?;

    // memmap dataset
    let mmap = unsafe { memmap2::Mmap::map(&std::fs::File::open("train_000.bin")?)? };
    let (_, data, _) = unsafe { mmap.align_to::<u16>() };

    let mut step = 0usize;
    let micro_bs = 32;
    let accum = 8;
    let seq_len = cfg.seq_len;
    let total_tokens = 2_000_000_000u64;
    let mut tokens_seen = 0u64;
    let start = std::time::Instant::now();

    while tokens_seen < total_tokens {
        let mut accum_loss = 0.0f32;
        opt.zero_grad()?;
        for _ in 0..accum {
            let batch = sample_batch(data, micro_bs, seq_len, &dev)?; // [B, L] u16 -> bf16 embed
            let logits = model.forward(&batch)?; // [B, L, vocab]
            let loss = candle_nn::loss::cross_entropy(&logits.flatten(0,1)?, &batch.flatten(0,1)?)?;
            let loss = (loss / accum as f64)?;
            loss.backward()?;
            accum_loss += loss.to_scalar::<f32>()?;
        }
        opt.step()?; // with clip 1.0, cosine decay
        tokens_seen += (micro_bs * seq_len * accum) as u64;
        step += 1;

        if step % 10 == 0 {
            let elapsed = start.elapsed().as_secs_f64();
            let tps = tokens_seen as f64 / elapsed;
            let mfu = (6.0 * 135e6 * tps) / 1.8e15; // vs 1800 TFLOPS BF16
            tracing::info!("step={} loss={:.3} tok/s={:.0} MFU={:.1}% vram={:.1}GB",
                step, accum_loss, tps, mfu*100.0, dev.mem_info()?.used / 1e9);
            if tps < 70000.0 { tracing::warn!("Below target throughput!") }
        }
        if step % 100 == 0 { model.save_safetensors(&format!("ckpt_{step}.safetensors"))?; }
    }
    Ok(())
}

    Adam m/v FP32 + master: ∼1.6 GB
