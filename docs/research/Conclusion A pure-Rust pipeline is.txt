### 5. Pre-training Loop & VRAM Management (`src/train/mod.rs`)

```rust
use candle_core::{Device, DType, Tensor};
use candle_nn::{AdamW, Optimizer, ParamsAdamW, loss::cross_entropy};
use std::time::Instant;

pub struct TrainConfig {
    pub micro_batch_size: usize,   // e.g. 8 or 16 sequences
    pub grad_accum_steps: usize,   // e.g. 4–8 → effective batch 32–128
    pub seq_len: usize,            // 2048
    pub max_steps: usize,
    pub lr: f64,                   // peak 2.5e-3
    pub warmup_steps: usize,       // 1000
    pub weight_decay: f64,         // 0.1
    pub beta1: f64,                // 0.9
    pub beta2: f64,                // 0.95
}

pub fn train(
    model: &mut Llama,
    tokens_mmap: &[u16],           // zero-copy from memmap2
    device: &Device,
    cfg: &TrainConfig,
) -> Result<()> {
    let dtype = DType::BF16; // or F8 if Candle exposes Blackwell FP8
    let mut opt = AdamW::new(
        /* all vars */,
        ParamsAdamW {
            lr: cfg.lr,
            beta1: cfg.beta1,
            beta2: cfg.beta2,
            eps: 1e-8,
            weight_decay: cfg.weight_decay,
        },
    )?;

    let mut step = 0usize;
    let start = Instant::now();
    let mut tokens_seen = 0u64;

    // Cosine schedule with linear warmup
    let lr_at = |s: usize| -> f64 {
        if s < cfg.warmup_steps {
            cfg.lr * (s as f64 / cfg.warmup_steps as f64)
        } else {
            let progress = (s - cfg.warmup_steps) as f64 / (cfg.max_steps - cfg.warmup_steps) as f64;
            cfg.lr * 0.5 * (1.0 + (std::f64::consts::PI * progress).cos())
        }
    };

    // Data loader: random or sequential windows of seq_len from mmap
    // For each micro-batch:
    //   construct input_ids / labels (shifted)
    //   forward → loss
    //   loss = loss / grad_accum_steps
    //   backward
    //   if (step+1) % grad_accum_steps == 0 { opt.step(); opt.zero_grad(); update lr }

    // Logging every N steps:
    //   tok/s = tokens_seen as f64 / start.elapsed().as_secs_f64()

    // Checkpoint periodically to safetensors
    Ok(())
```
